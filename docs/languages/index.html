Node, Python, PHP & Go

Nightjar needs no SDK. Any HTTP client works. Keep the URL in an environment variable.

Node.js

const URL = process.env.NIGHTJAR_URL;

export async function monitored(fn) {
  await fetch(`${URL}/start`).catch(() => {});
  try {
    const result = await fn();
    await fetch(URL).catch(() => {});
    return result;
  } catch (err) {
    await fetch(`${URL}/fail`, { method: "POST", body: String(err?.stack ?? err).slice(-8000) }).catch(() => {});
    throw err;
  }
}

await monitored(() => syncStripeInvoices());

The .catch(() => {}) calls make sure a monitoring hiccup never breaks the job itself.

Python

import os, traceback, urllib.request

URL = os.environ["NIGHTJAR_URL"]

def ping(path="", data=None):
    try:
        urllib.request.urlopen(URL + path, data=data, timeout=10)
    except Exception:
        pass  # never let monitoring break the job

ping("/start")
try:
    run_etl()
    ping()
except Exception:
    ping("/fail", traceback.format_exc()[-8000:].encode())
    raise

PHP / Laravel

Laravel’s scheduler has built-in hooks:

$schedule->command('reports:send')
    ->dailyAt('06:30')
    ->timezone('Europe/London')
    ->pingBefore(env('NIGHTJAR_URL').'/start')
    ->pingOnSuccess(env('NIGHTJAR_URL'))
    ->pingOnFailure(env('NIGHTJAR_URL').'/fail');

Go

func ping(suffix string, body io.Reader) {
	c := http.Client{Timeout: 10 * time.Second}
	if resp, err := c.Post(os.Getenv("NIGHTJAR_URL")+suffix, "text/plain", body); err == nil {
		resp.Body.Close()
	}
}

func main() {
	ping("/start", nil)
	if err := run(); err != nil {
		ping("/fail", strings.NewReader(err.Error()))
		os.Exit(1)
	}
	ping("", nil)
}
